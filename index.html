2. Perhatikan Kode JavaScript

Sebelum / Bug
function cari(nama, kata) {
  return nama.includes(kata);
}
Sesudah / Fix
function cari(nama, kata) {
  return nama.toLowerCase()
    .includes(kata.trim().toLowerCase());
}
